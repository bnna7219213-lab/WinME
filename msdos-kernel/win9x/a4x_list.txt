    16                              <1> ; --- dbg_puts : emit a NUL-terminated string literal to 0xE9 ----------------
    17                              <1> ;  Accepts 1 or more arguments (all collected into %1 as a single token list).
    18                              <1> ;  Usage:  dbg_puts "hello", 13, 10, 0
    19                              <1> ;  Works in BITS 16 and BITS 32 (detected via __BITS__).
    20                              <1> %macro dbg_puts 1+
    21                              <1>     jmp     %%skip
    22                              <1> %%str   db  %1
    23                              <1> %%skip:
    24                              <1> %if __BITS__ == 32
    25                              <1>     push    esi
    26                              <1>     mov     esi, %%str
    27                              <1> %%.loop:
    28                              <1>     lodsb
    29                              <1>     or      al, al
    30                              <1>     jz      %%.done
    31                              <1>     out     0xE9, al
    32                              <1>     jmp     %%.loop
    33                              <1> %%.done:
    34                              <1>     pop     esi
    35                              <1> %else

    37                              <1>     mov     si, %%str
    38                              <1> %%.loop:
    39                              <1>     lodsb
    40                              <1>     or      al, al
    41                              <1>     jz      %%.done
    42                              <1>     out     0xE9, al
    43                              <1>     jmp     %%.loop
    44                              <1> %%.done:

    58                              <1> %macro dbg_hex8 0

    61                              <1>     shr     al, 4
    62                              <1>     call    %%hex_nibble

    94                              <1>     add     al, 7            ; 'A' - '0' - 10 = 7
    95                              <1> %%ok2:

    97                              <1>     pop     eax
    98                              <1>     sub     ecx, 4

     1                              <1> ; ============================================================================
     2                              <1> ; gdt.inc — GDT (Global Descriptor Table) macros for 32-bit protected mode
     3                              <1> ; ----------------------------------------------------------------------------
     4                              <1> ;  Usage:  %include "common/gdt.inc"
     5                              <1> ;
     6                              <1> ;  Provides:
     7                              <1> ;    gdt_entry  base, limit, flags, access  — build one 8-byte descriptor
     8                              <1> ;    gdt_load   label                       — lgdt from a gdt_descriptor
     9                              <1> ;
    10                              <1> ;  Standard descriptor sets are provided as macros:
    11                              <1> ;    GDT_FLAT    — null + flat code + flat data (4GB each)
    12                              <1> ;    GDT_RING3   — flat + ring3 code + ring3 data (adds two entries)
    13                              <1> ; ============================================================================
    14                              <1> 
    15                              <1> ; --- Descriptor field layout (x86 GDT entry, 8 bytes) ---
    16                              <1> ;   bits  0-15  : limit  low
    17                              <1> ;   bits 16-39  : base   low/mid
    18                              <1> ;   bits 40-47  : access byte
    19                              <1> ;   bits 48-51  : limit  high (bits 16-19)
    20                              <1> ;   bits 52-55  : flags   (Granularity, Size, L, Reserved)
    21                              <1> ;   bits 56-63  : base   high
    22                              <1> 
    23                              <1> struc gdt_entry_t
    24 00000000 ????                <1>     .limit_lo   resw 1
    25 00000002 ????                <1>     .base_lo    resw 1
    26 00000004 ??                  <1>     .base_mid   resb 1
    27 00000005 ??                  <1>     .access     resb 1
    28 00000006 ??                  <1>     .flags_lim  resb 1
    29 00000007 ??                  <1>     .base_hi    resb 1
    30                              <1> endstruc
    31                              <1> 
    32                              <1> ; Access byte bits:
    33                              <1> ;   7 Present | 6-5 DPL | 4 S(1=code/data) | 3 E/DT | 2 DC | 1 R/W | 0 A
    34                              <1> ACCESS_RING0_CODE  equ 0x9A   ; Present, ring0, code, executable, read
    35                              <1> ACCESS_RING0_DATA  equ 0x92   ; Present, ring0, data, writable
    36                              <1> ACCESS_RING3_CODE  equ 0xFA   ; Present, ring3, code, executable, read
    37                              <1> ACCESS_RING3_DATA  equ 0xF2   ; Present, ring3, data, writable
    38                              <1> 
    39                              <1> ; Flags nibble (bits 52-55):
    40                              <1> ;   7 Granularity (1=4K) | 6 D/B (1=32-bit) | 5 L (64-bit, keep 0) | 4 Reserved
    41                              <1> FLAG_4K_32BIT      equ 0xCF   ; G=1, D=1, L=0, 0
    42                              <1> 
    43                              <1> ; --- gdt_entry : emit one 8-byte descriptor --------------------------------
    44                              <1> ;  %1=base(32)  %2=limit(20)  %3=access  %4=flags
    45                              <1> %macro gdt_entry 4
    46                              <1>     dw      (%2) & 0xFFFF                       ; limit 0-15
    47                              <1>     dw      (%1) & 0xFFFF                       ; base  0-15
    48                              <1>     db      ((%1) >> 16) & 0xFF                  ; base 16-23
    49                              <1>     db      (%3) & 0xFF                          ; access
    50                              <1>     db      (((%2) >> 16) & 0x0F) | ((%4) & 0xF0) ; flags + limit 16-19
    51                              <1>     db      ((%1) >> 24) & 0xFF                  ; base 24-31
    52                              <1> %endmacro
    53                              <1> 
    54                              <1> ; --- GDT descriptor (pseudo-descriptor for lgdt) ----------------------------
    55                              <1> struc gdt_desc_t
    56 00000000 ????                <1>     .limit   resw 1
    57 00000002 ????????            <1>     .base    resd 1
    58                              <1> endstruc

    69                              <1> ; --- GDT_FLAT : null + ring0 code (4GB) + ring0 data (4GB) ------------------
    70                              <1> ;  Use after entering 32-bit mode. Selector 0x08 = code, 0x10 = data.
    71                              <1> %macro GDT_FLAT 0
    72                              <1> ALIGN 8
    73                              <1> gdt_data:
    74                              <1>     gdt_entry 0, 0, 0, 0                       ; null descriptor (selector 0x00)
    75                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING0_CODE, FLAG_4K_32BIT  ; code  (selector 0x08)
    76                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING0_DATA, FLAG_4K_32BIT  ; data  (selector 0x10)
    77                              <1> gdt_descriptor:
    78                              <1>     dw      gdt_data_end - gdt_data - 1
    79                              <1>     dd      gdt_data
    80                              <1> gdt_data_end:
    81                              <1> %endmacro
    82                              <1> 
    83                              <1> ; --- GDT_RING3 : GDT_FLAT + ring3 code + ring3 data + TSS -------------------
    84                              <1> ;  Selector 0x08 = ring0 code, 0x10 = ring0 data,
    85                              <1> ;  0x18 = ring3 code, 0x20 = ring3 data, 0x28 = TSS
    86                              <1> %macro GDT_RING3 0
    87                              <1> ALIGN 8
    88                              <1> gdt_data:
    89                              <1>     gdt_entry 0, 0, 0, 0                       ; null     (0x00)
    90                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING0_CODE, FLAG_4K_32BIT  ; ring0 code (0x08)
    91                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING0_DATA, FLAG_4K_32BIT  ; ring0 data (0x10)
    92                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING3_CODE, FLAG_4K_32BIT  ; ring3 code (0x18)
    93                              <1>     gdt_entry 0, 0xFFFFF, ACCESS_RING3_DATA, FLAG_4K_32BIT  ; ring3 data (0x20)
    94                              <1>     ; TSS descriptor (placeholder, 104 bytes minimum)
    95                              <1>     gdt_entry 0, 0x67, 0x89, 0x40               ; TSS      (0x28) — present, ring0, 32-bit
    96                              <1> gdt_descriptor:
    97                              <1>     dw      gdt_data_end - gdt_data - 1
    98                              <1>     dd      gdt_data
    99                              <1> gdt_data_end:
   100                              <1> %endmacro
